TomoMod_SOV_develop_state = {
	if = {
		limit = { is_owned_by = SOV }
		if = {
			limit = { infrastructure < 3 }
			add_building_construction = {
				type = infrastructure
				level = 2
				instant_build = yes
			}
		}
		else_if = {
			limit = { infrastructure < 4 }
			if = {
				limit = { infrastructure > 2 }
				set_state_flag = SOV_develop_the_east_raised
			}
			add_building_construction = {
				type = infrastructure
				level = 1
				instant_build = yes
			}
		}
		add_dynamic_modifier = { modifier = SOV_develop_the_east_construction }
	}
}
