# TomoIHE port: copied from vanilla 1.19.3 common/scripted_triggers/SOV_scripted_triggers.txt (lines 139-184)
SOV_is_soviet_republic_tag_strict = {
	# TomoIHE port: vanilla lists BLR, UKR, KAZ, GEO, ... (Soviet republic/releasable tags). None of them exist in
	# this mod (common/country_tags only defines 36 tags), so the trigger can never be true; avoid unknown-tag errors.
	always = no
}

# TomoIHE port: copied from vanilla 1.19.3 common/scripted_triggers/SOV_scripted_triggers.txt (lines 186-238)
SOV_is_soviet_republic_tag_lax = {
	# TomoIHE port: vanilla also lists the Soviet republic tags (BLR, UKR, ... MEL, EST, LAT, LIT) which do not
	# exist in this mod; only the tags that exist in common/country_tags are kept.
	OR = {
		tag = POL
		tag = FIN
		tag = MON
	}
}
