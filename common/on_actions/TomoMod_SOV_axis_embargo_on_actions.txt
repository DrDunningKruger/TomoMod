on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = SOV }
				SOV = {
					every_other_country = {
						limit = {
							OR = {
								is_in_faction_with = GER
								original_tag = GER
								original_tag = BUL
								original_tag = YUG
							}
							NOT = { is_in_faction_with = SOV }
						}
						if = {
							limit = { NOT = { is_embargoing = SOV } }
							send_embargo = SOV
						}
						if = {
							limit = { NOT = { SOV = { is_embargoing = PREV } } }
							SOV = { send_embargo = PREV }
						}
					}
				}
			}
		}
	}
	on_monthly_SOV = {
		effect = {
			every_other_country = {
				limit = {
					OR = {
						is_in_faction_with = GER
						original_tag = GER
						original_tag = BUL
						original_tag = YUG
					}
					NOT = { is_in_faction_with = SOV }
				}
				if = {
					limit = { NOT = { is_embargoing = SOV } }
					send_embargo = SOV
				}
				if = {
					limit = { NOT = { SOV = { is_embargoing = PREV } } }
					SOV = { send_embargo = PREV }
				}
			}
		}
	}
}
