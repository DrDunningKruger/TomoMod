add_namespace = siam

country_event = {
	id = siam.10
	title = siam.10.t
	desc = siam.10.desc
	picture = GFX_report_event_SWE_psilander_affair 

	is_triggered_only = yes

	#Sent peace offer
	option = {
		name = siam.10.a
		ai_chance = {
			base = 1
		}

		if = {
			limit = {
				has_variable = SIA.limited_war_target
				var:SIA.limited_war_target = {
					exists = yes
				}
			}
			var:SIA.limited_war_target = { country_event = { id = siam.11 hours = 2 } }
		}
		else = {
			# Target already defeated/annexed - finish the limited war locally so it doesn't hang
			SIA_peace_out = yes
			set_rule = { can_join_factions = yes }
			SIA_re_add_to_old_faction = yes
			SIA_clear_limited_war_variables = yes
		}
	}

	#Continue the war as Siam
	option = {
		name = siam.10.b
		ai_chance = {
			base = 0
		}

		if = {
			limit = {
				has_variable = SIA.limited_war_target
				var:SIA.limited_war_target = {
					exists = yes
					has_capitulated = no
				}
			}
			var:SIA.limited_war_target = { country_event = { id = siam.14 hours = 2 } }
		}

		add_war_support = 0.1
		SIA_clear_limited_war_variables = yes
	}
}

country_event = {
	id = siam.12
	title = siam.12.t
	desc = siam.12.desc
	picture = GFX_report_event_SWE_psilander_affair 

	is_triggered_only = yes

	#End the war as Defender
	option = {
		name = siam.12.a
		ai_chance = {
			base = 1
		}

		SIA_peace_out = yes
		set_rule = { can_join_factions = yes } 
		for_each_scope_loop = {
			array =  limited_war_states
			owner = {
				if = {
					limit = {
						FROM = { NOT = { is_in_array = { array = lost_states_due_to_limited_war value = PREV.PREV } } }
					}
					FROM = { add_to_array = { array = lost_states_due_to_limited_war value = PREV.PREV } }
				}
			}
			transfer_state_to = ROOT

		}

		#Re-add Siam into their old faction if they were in a faction before
		SIA_re_add_to_old_faction = yes
		SIA_clear_limited_war_variables = yes

		hidden_effect = {
    		for_each_scope_loop = {
    			array = FROM.countries_that_guaranteed_me_pre_limited_war
    			FROM = {
    				diplomatic_relation = { 
    					country = PREV
    					relation = guarantee 
    					active = yes 
    				}
    			}
    		}

    		FROM = {
    			if = {
    				limit = { has_country_flag = SIA_limited_war_could_join_factions }
    				set_rule = { can_join_factions = yes }
    				clr_country_flag = SIA_limited_war_could_join_factions
    			}
    		}
		}
	}
}

country_event = {
	id = siam.16
	title = siam.16.t
	desc = siam.16.desc
	picture = GFX_report_event_SWE_psilander_affair 

	is_triggered_only = yes

	#Cede State, end war
	option = {
		name = siam.16.a
		ai_chance = {
			base = 1
		}
		
		if = {
			limit = {
				FROM = { 
					OR = {
						check_variable = { lost_states_due_to_limited_war^num > 0 }	
						any_allied_country = {
							check_variable = { lost_states_due_to_limited_war^num > 0 }	
						}
					}
				}
			}

			FROM = {
				if = {
					limit = {
						check_variable = { lost_states_due_to_limited_war^num > 0 }	
					}
					for_each_scope_loop = {
						array = lost_states_due_to_limited_war
						
						transfer_state_to = PREV
					}

					clear_array = lost_states_due_to_limited_war
				}
				every_allied_country = {
					if = {
						limit = {
							check_variable = { lost_states_due_to_limited_war^num > 0 }	
						}
						for_each_scope_loop = {
							array = lost_states_due_to_limited_war


							transfer_state_to = PREV
						}

						clear_array = lost_states_due_to_limited_war
					}
				}
			}
		}
		else = {
			add_stability = -0.3
			add_war_support = -0.3
			random_core_state = {
				limit = {
					arms_factory > 0
				}
				
				add_building_construction = {
					type = arms_factory
					level = -1
					instant_build = yes
				}
			}
		}
		
		SIA_peace_out = yes
		SIA_re_add_to_old_faction = yes
		SIA_clear_limited_war_variables = yes



	}

	#No we would rather die I guess
	option = {
		name = siam.16.b
		ai_chance = {
			base = 0
		}
		FROM = {
			country_event = { id = siam.17 hours = 2 }
		}


		add_timed_idea = {
			idea = idea_SIA_military_humiliation
			days = 780
		}

		SIA_re_add_to_old_faction = yes
		SIA_clear_limited_war_variables = yes
	}

}
